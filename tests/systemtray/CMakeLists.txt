project(systemtray_drag_tests LANGUAGES CXX)

find_package(Qt6 REQUIRED COMPONENTS Quick QuickTest)
find_package(KF6Config REQUIRED)

target_compile_definitions(systemtray-drag-tests PRIVATE QUICK_TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(systemtray-drag-tests PRIVATE TRAY_CONFIG_PATH="${CMAKE_CURRENT_SOURCE_DIR}/../../plasma/plasmoids/src/systemtray_src/systemtray/package/contents/config/main.xml")
enable_testing()
add_test(NAME systemtray-drag COMMAND systemtray-drag-tests)
set_tests_properties(systemtray-drag PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
